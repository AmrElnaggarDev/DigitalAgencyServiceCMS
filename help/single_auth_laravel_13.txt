<?php
namespace App\Http\Controllers;
use App\Mail\Websitemail;
use App\Models\User;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Facades\Mail;
use Illuminate\Support\Str;
class AuthController extends Controller
{
    public function showRegister()
    {
        return view('auth.register');
    }
    public function register(Request $request)
    {
        $request->validate([
            'name' => 'required|string|max:255',
            'email' => 'required|email|unique:users,email',
            'password' => 'required|string|min:6|confirmed',
        ]);
        $token = Str::random(64);
        $user = User::create([
            'name' => $request->name,
            'email' => $request->email,
            'password' => Hash::make($request->password),
            'verification_token' => $token,
        ]);
        $verify_url = route('verify', ['token' => $token]);
        $subject = 'Please verify your email';
        $message = '<h3>Hello '.$user->name.',</h3>';
        $message .= '<p>Thanks for registering. Please click the link below to verify your email address:</p>';
        $message .= '<p><a href="'.$verify_url.'">'.$verify_url.'</a></p>';
        Mail::to($user->email)->send(new Websitemail($subject, $message));
        return redirect()->route('login')->with('success', 'Registered successfully. Please check your email to verify your account.');
    }
    public function verify($token)
    {
        $user = User::where('verification_token', $token)->first();
        if (! $user) {
            return redirect()->route('login')->with('error', 'Invalid or expired verification link.');
        }
        $user->email_verified_at = now();
        $user->verification_token = null;
        $user->save();
        return redirect()->route('login')->with('success', 'Email verified successfully. You can now log in.');
    }
    public function showLogin()
    {
        return view('auth.login');
    }
    public function login(Request $request)
    {
        $request->validate([
            'email' => 'required|email',
            'password' => 'required|string',
        ]);
        $user = User::where('email', $request->email)->first();
        if (! $user || ! Hash::check($request->password, $user->password)) {
            return back()->with('error', 'Invalid email or password.')->withInput($request->only('email'));
        }
        if (is_null($user->email_verified_at)) {
            return back()->with('error', 'Please verify your email before logging in.')->withInput($request->only('email'));
        }
        Auth::login($user, $request->boolean('remember'));
        $request->session()->regenerate();
        return redirect()->route('dashboard');
    }
    public function logout(Request $request)
    {
        Auth::logout();
        $request->session()->invalidate();
        $request->session()->regenerateToken();
        return redirect()->route('login')->with('success', 'Logged out successfully.');
    }
    public function dashboard()
    {
        return view('dashboard');
    }
    public function showForgotPassword()
    {
        return view('auth.forgot-password');
    }
    public function sendResetLink(Request $request)
    {
        $request->validate([
            'email' => 'required|email',
        ]);
        $user = User::where('email', $request->email)->first();
        if (! $user) {
            return back()->with('error', 'No account found with that email address.')->withInput();
        }
        $token = Str::random(64);
        $user->reset_token = $token;
        $user->save();
        $reset_url = route('password.reset', ['token' => $token]).'?email='.urlencode($user->email);
        $subject = 'Reset your password';
        $message = '<h3>Hello '.$user->name.',</h3>';
        $message .= '<p>We received a request to reset your password. Click the link below to set a new password:</p>';
        $message .= '<p><a href="'.$reset_url.'">'.$reset_url.'</a></p>';
        $message .= '<p>If you did not request a password reset, you can ignore this email.</p>';
        Mail::to($user->email)->send(new Websitemail($subject, $message));
        return back()->with('success', 'A password reset link has been sent to your email.');
    }
    public function showResetPassword(Request $request, $token)
    {
        return view('auth.reset-password', [
            'token' => $token,
            'email' => $request->query('email'),
        ]);
    }
    public function resetPassword(Request $request)
    {
        $request->validate([
            'token' => 'required|string',
            'email' => 'required|email',
            'password' => 'required|string|min:6|confirmed',
        ]);
        $user = User::where('email', $request->email)
            ->where('reset_token', $request->token)
            ->first();
        if (! $user) {
            return back()->with('error', 'Invalid or expired password reset link.')->withInput($request->only('email'));
        }
        $user->password = Hash::make($request->password);
        $user->reset_token = null;
        $user->save();
        return redirect()->route('login')->with('success', 'Password has been reset. You can now log in with your new password.');
    }
}

<?php
namespace App\Mail;
use Illuminate\Bus\Queueable;
use Illuminate\Mail\Mailable;
use Illuminate\Mail\Mailables\Content;
use Illuminate\Mail\Mailables\Envelope;
use Illuminate\Queue\SerializesModels;
class Websitemail extends Mailable
{
    use Queueable, SerializesModels;
    public $subject, $body;
    public function __construct($subject, $body)
    {
        $this->subject = $subject;
        $this->body = $body;
    }
    public function envelope(): Envelope
    {
        return new Envelope(
            subject: $this->subject,
        );
    }
    public function content(): Content
    {
        return new Content(
            view: 'email',
            with: [
                'subject' => $this->subject,
                'body' => $this->body,
            ],
        );
    }
}

========================================
✅ 3 = Token Hashing in DB
========================================
👉 Problem:
Now token is saved in DB as plain text format. So, if any person gets access to read the DB, he will get all the valid tokens.

👉 Fix: During save, use hash('sha256', $token), during verify use hash to compare. Plain token go to email and hash will go to DB.

👉 Why we will use SHA-256 and not bcrypt?
bcrypt or Hash::make = It is slow (intentional, for password) and each time it will give you separate hash. So in the DB lookup, you can not use where() clause.
SHA-256 = It is fast, deterministic. Same input = Same output. So you can use the direct look up in the indexed column.
Str::random(64) = 64 chars random. In this method, entropy is large. So rainbow table attack is not possible. SHA-256 is enough.

👉 Change this:
'verification_token' => $token,
To:
'verification_token' => hash('sha256', $token),

and Change this:
$user = User::where('verification_token', $token)->first();
To:
$user = User::where('verification_token', hash('sha256', $token))->first();

and Change this:
$user->reset_token = $token;
To:
$user->reset_token = hash('sha256', $token);

and Change this:
->where('reset_token', $request->token)
To:
->where('reset_token', hash('sha256', $request->token))

========================================
✅ 5 = Timing Attack on Login
========================================
👉 Problem:
If User::where('email', ...)->first() is false, then Hash::check() will not be called. In that time the response time will be very fast and the attacker will easily understand if email exists or not.

👉 Fix:
So, to confuse the attacker, we will use Hash::check() function which is slow and takes more time than usual.

    ...
}

$passwordHash = $user ? $user->password : $dummyHash;
$validPassword = Hash::check($request->password, $passwordHash);
if (! $user || ! $validPassword)
{
    ...
}

========================================
✅ 7 = HTTPS in Production
========================================
👉 Problem: Session, cookie and passing data into the cookie
In .env, there is:
APP_URL=http://localhost

In production if you keep http or put https, but pass cookie without secure flag, then: 

1 = Session can be Hijacked
2 = Password can be passed as plain text
3 = Reset password token can be leaked
4 = CSRF token and XSS protection can be weak

1 = Session Hijacking
After a successful Laravel login, the browser stores a cookie such as:
laravel_session=eyJpdiI6Ik9z...; expires=...; path=/
This cookie proves that the user is authenticated. The browser automatically sends it with every request.
> Where is the problem?
- When a website uses plain HTTP, all data travels across the network in plain text. If someone can monitor the traffic (which is especially common on public Wi-Fi), they can capture this cookie and reuse it in their own browser. Once they do that, they can access the victim's account without knowing the password.
> Who can see it?
- Another user on the same public Wi-Fi network (e.g., in a café or airport) using tools like Wireshark.
- A compromised router or ISP.
- A Man-in-the-Middle (MITM) attacker running a fake Wi-Fi hotspot.
> Demo Scenario
- A user logs in to your website while connected to a café's public Wi-Fi using HTTP.
- A nearby attacker is running Wireshark and monitoring network traffic.
- The attacker sees a request containing:
Cookie: laravel_session=xyz123
- The attacker copies the cookie value, open their browser, go to: DevTools → Application → Cookies, and add the stolen cookie.
- They refresh your website. 
- The attacker is now logged in as the victim without ever knowing the password.

2 = Password can be passed as plain text
When a user submits the login form, the password is sent in the body of the POST request:
POST /login HTTP/1.1
Host: example.com
email=user@x.com&password=MySecret123!
If the website is using plain HTTP, the entire request body is transmitted as plain text across the network. An attacker only needs to see the traffic once to obtain the user's password.
As a result, all the effort spent securely hashing passwords in the database (for example, using bcrypt with 12 rounds) becomes meaningless, because the original password is exposed while traveling over the network.

3 = Reset password token can be leaked
Your reset link is something like this:
http://example.com/reset-password/abc123xyz...?email=user@x.com
If attacker can see token in http, then if you keep hash in DB or not, it is meaningless. Because he got the original token and he can hit the URL and reset the password.

4 = CSRF token and XSS protection can be weak
If you send CSRF token as plain text, attacker can re-use that.
Browser security features (ex: Secure-only cookies, HSTS) will not work in HTTP.

👉 Fix: 2 layers are needed
A. Enable HTTPS in your server.
It will encrypt the full traffic. 
B. Mark the Cookie as "Secure-only"
Browser will send cookie only in HTTPS connection, not in HTTP.
Our task is only to work with option B. Option A will be managed in server. 

👉 Our Procedure:
We will use SESSION_SECURE_COOKIE in session.php and .env in such way, so that: 
- In Local dev (HTTP), cookie will work normally. (SESSION_SECURE_COOKIE=false)
- In Production (HTTPS), cookie will be passed in only HTTPS. (SESSION_SECURE_COOKIE=true)

👉 We will add 2 more hardenings:
- SESSION_HTTP_ONLY=true : You can not read cookie from javascript (XSS Protection)
- SESSION_SAME_SITE=lax : Cookie will not go in cross-site request (CSRF more strong)
REMEMBER: This fix will not enable HTTPS itself. You have to do that on your hosting. But using this fix, it will ensure that, if any person tries to access the HTTP (in spite of having HTTPS), this will not send session cookie to him. So hijacking will be impossible. 

👉 .env: 
SESSION_SECURE_COOKIE=false
SESSION_HTTP_ONLY=true
SESSION_SAME_SITE=lax

👉 Run:
php artisan config:clear

👉 After uploading the files in web server:
SESSION_SECURE_COOKIE=true
APP_URL=https://yourdomain.com
APP_ENV=production
APP_DEBUG=false
